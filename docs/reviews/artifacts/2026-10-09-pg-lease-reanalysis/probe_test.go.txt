func TestLeaseReanalysisIsolationMatrix(t *testing.T) {
	for _, tc := range []struct {
		name      string
		isolation pgx.TxIsoLevel
		rollback  bool
	}{
		{name: "pool"},
		{name: "read_committed", isolation: pgx.ReadCommitted},
		{name: "repeatable_read", isolation: pgx.RepeatableRead},
		{name: "serializable", isolation: pgx.Serializable},
		{name: "winner_rollback", rollback: true},
	} {
		t.Run(tc.name, func(t *testing.T) {
			ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
			defer cancel()
			poolA, poolB := newIntegrationLeasePools(t)
			winnerTx, err := poolA.Begin(ctx)
			if err != nil {
				t.Fatal(err)
			}
			defer winnerTx.Rollback(ctx)
			winnerStore, err := NewPGLeaseStore(winnerTx)
			if err != nil {
				t.Fatal(err)
			}
			now := time.Now().UnixMilli()
			winner, acquired, err := winnerStore.TryAcquire(ctx, 12, "owner-a", now, 10_000, now+10_000, 1_000)
			if err != nil || !acquired {
				t.Fatalf("winner: acquired=%v err=%v", acquired, err)
			}
			var pid int32
			if err := winnerTx.QueryRow(ctx, "SELECT pg_backend_pid()").Scan(&pid); err != nil {
				t.Fatal(err)
			}
			var loserDB LeaseDB = poolB
			if tc.isolation != "" {
				tx, err := poolB.BeginTx(ctx, pgx.TxOptions{IsoLevel: tc.isolation})
				if err != nil {
					t.Fatal(err)
				}
				defer tx.Rollback(ctx)
				loserDB = tx
			}
			loserStore, err := NewPGLeaseStore(loserDB)
			if err != nil {
				t.Fatal(err)
			}
			type result struct {
				state    LeaseState
				acquired bool
				err      error
			}
			results := make(chan result, 1)
			go func() {
				state, acquired, err := loserStore.TryAcquire(ctx, 12, "owner-b", now, 10_000, now+10_000, 1_000)
				results <- result{state, acquired, err}
			}()
			waitForLeaseConflictWait(t, poolA, pid)
			if tc.rollback {
				err = winnerTx.Rollback(ctx)
			} else {
				err = winnerTx.Commit(ctx)
			}
			if err != nil {
				t.Fatal(err)
			}
			var loser result
			select {
			case loser = <-results:
			case <-ctx.Done():
				t.Fatal(ctx.Err())
			}
			if tc.isolation == pgx.RepeatableRead || tc.isolation == pgx.Serializable {
				var pgErr *pgconn.PgError
				if loser.acquired || !errors.As(loser.err, &pgErr) || pgErr.Code != "40001" {
					t.Fatalf("want serialization failure, got acquired=%v state=%+v err=%v", loser.acquired, loser.state, loser.err)
				}
				t.Logf("acquired=false SQLSTATE=%s", pgErr.Code)
				return
			}
			if loser.err != nil {
				t.Fatal(loser.err)
			}
			if tc.rollback {
				if !loser.acquired || loser.state.OwnerID != "owner-b" {
					t.Fatalf("rollback contender: %+v", loser)
				}
				assertPersistedLeaseOwnerAndBounds(t, poolA, 12, "owner-b", loser.state.ReservedUntilMillis, loser.state.GenerationFenceMillis)
			} else if loser.acquired || loser.state.NodeID != 12 || loser.state.OwnerID != "owner-a" || loser.state.ReservedUntilMillis != winner.ReservedUntilMillis || loser.state.GenerationFenceMillis != winner.GenerationFenceMillis {
				t.Fatalf("committed winner mismatch: %+v winner=%+v", loser, winner)
			}
			t.Logf("acquired=%v owner=%s", loser.acquired, loser.state.OwnerID)
		})
	}
}

func TestLeaseReanalysisFallbackMapping(t *testing.T) {
	for _, noRows := range []bool{false, true} {
		name := "success"
		if noRows {
			name = "no_rows"
		}
		t.Run(name, func(t *testing.T) {
			db := &fakeLeaseDB{queryRows: []pgx.Row{
				fakeLeaseRow{scan: func(dest ...any) error {
					*(dest[0].(*bool)) = false
					*(dest[1].(*int)) = 7
					*(dest[5].(*int64)) = 2_000
					*(dest[6].(*string)) = "retry"
					return nil
				}},
				fakeLeaseRow{scan: func(dest ...any) error {
					if noRows {
						return pgx.ErrNoRows
					}
					*(dest[0].(*int)) = 7
					*(dest[1].(*string)) = "owner-a"
					*(dest[2].(*int64)) = 3_000
					*(dest[3].(*int64)) = 3_200
					*(dest[4].(*int64)) = 2_200
					return nil
				}},
			}}
			store, err := NewPGLeaseStore(db)
			if err != nil {
				t.Fatal(err)
			}
			state, acquired, err := store.TryAcquire(context.Background(), 7, "owner-b", 2_000, 1_000, 3_000, 1_000)
			if err != nil || acquired || state.NodeID != 7 {
				t.Fatalf("state=%+v acquired=%v err=%v", state, acquired, err)
			}
			if noRows {
				if state.OwnerID != "" || state.ReservedUntilMillis != 0 || state.GenerationFenceMillis != 0 || state.DatabaseNowMillis != 2_000 {
					t.Fatalf("retry state=%+v", state)
				}
			} else if state.OwnerID != "owner-a" || state.ReservedUntilMillis != 3_000 || state.GenerationFenceMillis != 3_200 || state.DatabaseNowMillis != 2_200 {
				t.Fatalf("mapped state=%+v", state)
			}
			if len(db.queryArgs) != 1 || db.queryArgs[0] != 7 {
				t.Fatalf("readback node args=%v", db.queryArgs)
			}
			t.Logf("state=%+v acquired=%v", state, acquired)
		})
	}
}

func TestLeaseReanalysisFallbackWaitCharacterization(t *testing.T) {
	clock := &fakeClock{now: 2_000}
	stop := errors.New("stop after observing first retry")
	db := &fakeLeaseDB{queryRows: []pgx.Row{
		fakeLeaseRow{scan: func(dest ...any) error {
			clock.now += 200
			clock.monotonic += 200
			*(dest[0].(*bool)) = false
			*(dest[1].(*int)) = 7
			*(dest[5].(*int64)) = 2_000
			*(dest[6].(*string)) = "retry"
			return nil
		}},
		fakeLeaseRow{scan: func(dest ...any) error {
			*(dest[0].(*int)) = 7
			*(dest[1].(*string)) = "owner-a"
			*(dest[2].(*int64)) = 3_000
			*(dest[3].(*int64)) = 3_000
			*(dest[4].(*int64)) = 2_200
			return nil
		}},
		fakeLeaseRow{scan: func(...any) error { return stop }},
	}}
	store, err := NewPGLeaseStore(db)
	if err != nil {
		t.Fatal(err)
	}
	manager, err := NewLeaseManager(store, clock, LeaseManagerConfig{NodeID: 7, OwnerID: "owner-b", LeaseWindow: time.Second, AcquireTimeout: 2 * time.Second})
	if err != nil {
		t.Fatal(err)
	}
	_, err = manager.Acquire(context.Background())
	if !errors.Is(err, stop) {
		t.Fatalf("unexpected stop: %v", err)
	}
	if len(clock.durationSleeps) != 1 || clock.durationSleeps[0] != 600*time.Millisecond {
		t.Fatalf("sleeps=%v", clock.durationSleeps)
	}
	t.Log("readback remaining=800ms; manager sleep=600ms after 200ms conflict wait; retry occurs 200ms before lease expiry")
}
